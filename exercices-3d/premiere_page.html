(function () { var scene = document.querySelector('.scene'); var room = document.querySelector('.room'); var maxAngleY = 25; var maxAngleX = 6; scene.addEventListener('mousemove', function (e) { var rect = scene.getBoundingClientRect(); var xPercent = (e.clientX - rect.left) / rect.width - 0.5; var yPercent = (e.clientY - rect.top) / rect.height - 0.5; var angleY = xPercent * maxAngleY * 2; var angleX = -yPercent * maxAngleX * 2; room.style.transform = 'rotateX(' + angleX + 'deg) rotateY(' + angleY + 'deg)'; }); scene.addEventListener('touchmove', function (e) { e.preventDefault(); var touch = e.touches[0]; var rect = scene.getBoundingClientRect(); var xPercent = (touch.clientX - rect.left) / rect.width - 0.5; var yPercent = (touch.clientY - rect.top) / rect.height - 0.5; var angleY = xPercent * maxAngleY * 2; var angleX = -yPercent * maxAngleX * 2; room.style.transform = 'rotateX(' + angleX + 'deg) rotateY(' + angleY + 'deg)'; }, { passive: false }); scene.addEventListener('mouseleave', function () { room.style.transform = 'rotateX(0deg) rotateY(0deg)'; }); })();
